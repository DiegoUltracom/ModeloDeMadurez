-- creación de tablas CREATE TABLE Granero ( GraneroID INT PRIMARY KEY, Nombre VARCHAR(100), Ubicacion VARCHAR(100) ); CREATE TABLE Producto ( ProductoID INT PRIMARY KEY, Nombre VARCHAR(100), Precio DECIMAL(10, 2), Cantidad INT, GraneroID INT, FOREIGN KEY (GraneroID) REFERENCES Granero(GraneroID) ); -- Insertar en la tabla Granero INSERT INTO Granero (GraneroID, Nombre, Ubicacion) VALUES (1, 'Granero talanga', 'calle principal 123'), (2, 'Granero valle grande', 'avenida central 456'); -- Insertar en la tabla Producto INSERT INTO Producto (ProductoID, Nombre, Precio, Cantidad, GraneroID) VALUES (1, 'Maíz', 5.99, 100, 1), (2, 'Trigo', 6.99, 150, 1), (3, 'Arroz', 3.49, 200, 2), (4, 'Avena', 4.29, 120, 2), (5, 'Cebada', 4.99, 180, 2); -- 4 Preguntas con respuestas usando SELECT -- 1. Mostrar todos los graneros y sus productos SELECT g.Nombre AS Nombre_Granero, p.Nombre AS Nombre_Producto, p.Precio, p.Cantidad FROM Granero g JOIN Producto p ON g.GraneroID = p.GraneroID; -- 2. Mostrar el total de productos en cantidad en cada granero SELECT g.Nombre AS Nombre_Granero, SUM(p.Cantidad) AS Total_Cantidad FROM Granero g JOIN Producto p ON g.GraneroID = p.GraneroID GROUP BY g.Nombre; -- 3. Mostrar los productos con un precio superior a $5.00 SELECT Nombre, Precio FROM Producto WHERE Precio > 5.00; -- 4. Mostrar el granero que tiene más productos en cantidad SELECT g.Nombre AS Nombre_Granero, SUM(p.Cantidad) AS Total_Cantidad FROM Granero g JOIN Producto p ON g.GraneroID = p.GraneroID GROUP BY g.Nombre ORDER BY Total_Cantidad DESC LIMIT 1; -- 4 Modificaciones a los datos (Update-Delete) -- 1. Actualizar el precio del producto con ID 3 (Arroz) UPDATE Producto SET Precio = 3.99 WHERE ProductoID = 3; -- 2. Eliminar el producto con ID 5 (Cebada) DELETE FROM Producto WHERE ProductoID = 5; -- 3. Actualizar la ubicación del Granero B UPDATE Granero SET Ubicacion = 'Calle Nueva 722' WHERE GraneroID = 2; -- 4. Eliminar el granero con ID 1 (Granero talanga) y sus productos asociados DELETE FROM Granero WHERE GraneroID = 1;
